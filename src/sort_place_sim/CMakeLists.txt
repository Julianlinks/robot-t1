cmake_minimum_required(VERSION 3.16)
project(sort_place_sim)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sort_place_interfaces REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(cell_vacuum SHARED src/cell_vacuum.cpp)
ament_target_dependencies(cell_vacuum rclcpp sort_place_interfaces)
target_link_libraries(cell_vacuum gz-sim8::gz-sim8 gz-plugin2::register)
install(TARGETS cell_vacuum LIBRARY DESTINATION lib)
install(DIRECTORY launch config urdf worlds rviz DESTINATION share/${PROJECT_NAME})
ament_package()
